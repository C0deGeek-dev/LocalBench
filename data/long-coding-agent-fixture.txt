function Resolve-LlamaCppAllowedKvTypes {
    param(
        [string]$Mode,
        [hashtable]$KvTypes
    )

    $values = @('q8_0')
    if ($KvTypes.Contains('K')) { $values += @($KvTypes.K) }
    if ($Mode -eq 'turboquant') {
        if ($KvTypes.Contains('V')) { $values += @($KvTypes.V) }
        $values += @('turbo3', 'turbo4')
    }

    return @($values | Where-Object { $_ } | Select-Object -Unique)
}

function Invoke-LlamaCppTrialServer {
    param(
        [string]$Mode,
        [hashtable]$Overrides,
        [int]$Runs = 3
    )

    $buildParams = @{
        Mode = $Mode
        Port = $port
        ModelArgPath = $ModelArgPath
    }

    foreach ($k in @('KvK','KvV','NCpuMoe','SpecType','SpecDraftNMax')) {
        if ($Overrides.Contains($k) -and $null -ne $Overrides[$k]) {
            $buildParams[$k] = $Overrides[$k]
        }
    }

    $args = Build-LlamaServerArgs @buildParams
    Start-Process -FilePath $serverExe -ArgumentList $args

    foreach ($i in 1..$Runs) {
        $payload = @{
            prompt = $script:LlamaCppTunerPrompt
            n_predict = $script:LlamaCppTunerNPredict
            cache_prompt = $false
        }
        $resp = Invoke-RestMethod -Method Post -Uri "http://127.0.0.1:$port/completion" -Body ($payload | ConvertTo-Json)
        $responseText = [string]$resp.content
        $qualityOk = Test-LlamaCppOutputQuality -Text $responseText -MinChars 80 -MinWords 18
        if (-not $qualityOk) { $startupOk = $false }
    }
}

llama.cpp MTP notes from local help and upstream examples:

- The mtpturbo binary supports --spec-type with values including mtp.
- The same binary supports KV cache types q8_0, f16, turbo2, turbo3, and turbo4.
- MTP testing should use a single slot for deterministic server trials: --parallel 1.
- MoE CPU offload can prevent OOM, but too much GPU residency may still fail at model load time.
- A server that returns HTTP 200 and emits timing data can still be a bad trial if the generated content is nonsense.
